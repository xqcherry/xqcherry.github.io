**暴力 DP：$O(n^2)$**

对于每个 `i` 判断 `[i + minJ, i + maxJ]` 中 `f[j]` 可不可达

```
class Solution {
public:
    bool canReach(string s, int minJump, int maxJump) {
        int n = s.size();
        vector<int> f(n, 0);

        f[0] = 1;

        for(int i = 0; i < n; i ++ ) {
            if(!f[i]) continue;

            for(int j = i + minJump; j <= min(n - 1, i + maxJump); j ++ ) {
                if(s[j] == '0') f[j] = 1;
            }
        }

        return f[n - 1];
    }
};
```

**滑动窗口优化：$O(n)$**

对于每个当前的 `i` ，判断 `[j - maxJ, j - mixJ]` 中 有没有存在 `f[j] == 1` ，从 `f[j]` 跳过来
`[j - maxJ, j - mixJ]`是不断向右的滑动窗口，维护左右边界符不符合 `check`

```
class Solution {
public:
    bool canReach(string s, int minJump, int maxJump) {
        int n = s.size();

        vector<int> f(n, 0);
        f[0] = 1;

        int cnt = 0;
        for(int i = 1; i < n; i ++ ) {

            int r = i - minJump;
            if(r >= 0 && f[r]) cnt ++;

            int l = i - maxJump - 1;
            if(l >= 0 && f[l]) cnt --;

            if(s[i] == '0' && cnt) f[i] = 1;
        }

        return f[n - 1];
    }
};
```
