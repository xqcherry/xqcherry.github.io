```
class Solution {
public:
    vector<vector<int>> g, res;
    vector<int> path;
    int n ;

    vector<vector<int>> allPathsSourceTarget(vector<vector<int>>& graph) {

        g = graph;
        n = graph.size();

        path.push_back(0);
        dfs(0);

        return res;
    }

    void dfs(int u) {

        if(u == n - 1) {
            res.push_back(path);
            return ;
        }

        for(auto& v : g[u]) {
            path.push_back(v);
            dfs(v);
            path.pop_back();
        }
    }
};
```
