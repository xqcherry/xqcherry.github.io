```
#include<bits/stdc++.h>
using namespace std;

const int N = 10;

int n, res;
int g[N][N];

int check(int r, int c) {

    for(int i = r - 1; i >= 0; i --) {
        if(g[i][c] == 1) return 0;
    }

    for(int i = r - 1, j = c - 1; i >= 0 && j >= 0; i --, j --) {
        if(g[i][j] == 1) return 0;
    }

    for(int i = r - 1, j = c + 1; i >= 0 && j < n; i --, j ++) {
        if(g[i][j] == 1) return 0;
    }

    return 1;
}

void dfs(int r) {

    if(r == n) {
        res ++;
        return ;
    } 

    for(int c = 0; c < n; c ++ ) {
        if(check(r, c)) {
            g[r][c] = 1;
            dfs(r + 1);
            g[r][c] = -1;
        }
    }
}

int main() {

    cin >> n;

    for(int i = 0; i < n; i ++)
        for(int j = 0; j < n; j ++)
            g[i][j] = -1;

    dfs(0);

    cout << res << "\n";

    return res;
}
```
