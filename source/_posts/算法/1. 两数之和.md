> https://leetcode.cn/problems/two-sum/description/

```
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        int n = nums.size();

        map<int, int> m;
        vector<int> ans;

        for(int i = 0; i < n; i ++ ) m[nums[i]] = i;

        for(int i = 0; i < n; i ++ ) {
            auto x = nums[i];
            int k = target - x;
            if(m.count(k) && m[k] != i) {
                ans.push_back(i);
                ans.push_back(m[k]);
                break;
            }
        }

        return ans;
    }
};
```
