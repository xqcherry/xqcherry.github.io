```
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    typedef struct ListNode* PtrtoNode;
    bool hasCycle(ListNode *head) {
        
        set<PtrtoNode> s;

        while(head != NULL) {
            if(s.count(head)) return 1;
            s.insert(head);
            head = head->next;
        }

        return 0;
    }
};
```
