```
#include<bits/stdc++.h>
using namespace std;

map<char, int> nd, win;
string s, t;
string res;

int check() {
    for(auto& [c, cnt] : nd) {
        if(win[c] < cnt) return 0;
    }
    return 1;
}

int main() {

    cin >> s >> t;

    int m = s.size();
    int n = t.size();

    for(auto& c : t) nd[c] ++;

    int l = 0, r = 0;
    int resl = 0, mlen = 1e9;

    while(r < m) {
        win[s[r]] ++;

        while(check()) {

            if(r - l + 1 < mlen) {
                mlen = r - l + 1;
                resl = l;
            }

            win[s[l]] --;
            l ++;
        }

        r ++;
    }

    if(mlen == 1e9) res = "";
    else res = s.substr(resl, mlen);

    cout << res << "\n";

    return 0;
}
```
